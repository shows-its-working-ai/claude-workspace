<p class="note">Each dot is one measurement, on log scales. They lie close to a straight line, and its slope gives the
coast's fractal dimension D: here the fit gives @@D@@. The island is built so that the exact answer is
log 4 / log 3 = 1.262.</p>

<h2>What's going on</h2>
<p>Open the dividers to a fixed width, put one point on the coast, and step round: each step lands on the coast exactly
one ruler further on. Count the steps, multiply by the ruler, and you have a length. With a long ruler you step across
every bay. With a short one you follow the bays, then the coves inside the bays, and the coast gets longer.</p>
<p>For a smooth shape this stops: shorten the ruler enough and the length settles down. For this island it never
does, because every stretch of coast has smaller bumps on it, all the way down. Every time the ruler gets three times
shorter, the coast gets exactly four thirds longer. With the shortest ruler here (1.23 km, the size of the smallest
bump) it measures @@LMAX@@ km. Halving the ruler makes it about 2<sup>0.262</sup> = 1.20 times longer <i>on average</i>: it isn't steady. This island is built in thirds, so a ruler that falls between two of its sizes can come out shorter than a slightly longer one (try 84 km against 100 km). Only the trend is reliable, which is why the dots are fitted with a line.</p>
<h2>The real thing</h2>
<p>Lewis Fry Richardson noticed this while studying wars and borders: Portugal and Spain gave the length of their
shared border as 987 km and 1,214 km. In 1967 Benoit Mandelbrot wrote it up as "How Long Is the Coast of Britain?"
(<i>Science</i> 156, pages 636&ndash;638), describing real coasts by a number D. He gave about 1.25 for the west coast
of Britain, and about 1.02 for South Africa's much smoother one. Measured with a 100 km ruler, Britain's coast comes
to about 2,800 km; with a 50 km ruler, about 3,400 km. That's 1.21 times longer for half the ruler, almost what this
island does (1.20).</p>
<p class="note">How it was checked: a separate program builds the same island (its bumps point in or out at random,
and are re-rolled wherever the coast would cross itself), walks the dividers, and confirms that every ruler of
1/3, 1/9, ... of a side gives exactly 4/3, 16/9, ... times the side; the page's own walk matches it step for step.
The real-world numbers are from the Wikipedia article <a href="https://en.wikipedia.org/wiki/Coastline_paradox">Coastline paradox</a>, checked when I wrote this. Before
running anything I guessed the random island wouldn't cross itself. It did, until the re-rolling was added.</p>

<script>
const DATA = JSON.parse(document.getElementById('data').textContent), PTS = DATA.pts, SIDE = 300, $ = id => document.getElementById(id);
const RMAX = SIDE / 3, RMIN = SIDE / 243;
/* the divider walk: the same rule as coast.py's walk() */
function walk(r){
  let cur = PTS[0], i = 0; const seen = [cur];
  for (;;){
    let found = null;
    for (let k = i; k < PTS.length - 1; k++){
      const a = PTS[k], b = PTS[k + 1], dx = b[0] - a[0], dy = b[1] - a[1], fx = a[0] - cur[0], fy = a[1] - cur[1];
      const A = dx * dx + dy * dy, B = 2 * (fx * dx + fy * dy), C = fx * fx + fy * fy - r * r, disc = B * B - 4 * A * C;
      if (disc < 0) continue;
      const q = Math.sqrt(disc), t0 = k > i ? 0 : ((cur[0] - a[0]) * dx + (cur[1] - a[1]) * dy) / A;
      const ts = [(-B - q) / (2 * A), (-B + q) / (2 * A)].filter(t => t >= -1e-9 && t <= 1 + 1e-9).sort((x, y) => x - y)
        .filter(t => t > t0 + 1e-9).map(t => Math.min(Math.max(t, 0), 1));
      if (ts.length){ found = [k, [a[0] + ts[0] * dx, a[1] + ts[0] * dy]]; break; }
    }
    if (!found){ const last = PTS[PTS.length - 1]; return {steps: seen.length - 1, gap: Math.hypot(cur[0] - last[0], cur[1] - last[1]), seen}; }
    [i, cur] = found; seen.push(cur);
  }
}
const measure = rUnits => { const w = walk(rUnits); return {...w, L: w.steps * rUnits + w.gap}; };
const css = v => getComputedStyle(document.documentElement).getPropertyValue(v).trim();
const hist = [];
function drawMap(w){
  const c = $('map'), g = c.getContext('2d'), W = c.width, pad = 40, xs = PTS.map(p => p[0]), ys = PTS.map(p => p[1]);
  const x0 = Math.min(...xs), x1 = Math.max(...xs), y0 = Math.min(...ys), y1 = Math.max(...ys), s = (W - 2 * pad) / Math.max(x1 - x0, y1 - y0);
  const X = p => pad + (p[0] - x0) * s + ((W - 2 * pad) - (x1 - x0) * s) / 2, Y = p => W - pad - (p[1] - y0) * s - ((W - 2 * pad) - (y1 - y0) * s) / 2;
  g.clearRect(0, 0, W, W); g.beginPath(); PTS.forEach((p, k) => k ? g.lineTo(X(p), Y(p)) : g.moveTo(X(p), Y(p)));
  g.fillStyle = css('--land'); g.fill(); g.strokeStyle = css('--shore'); g.lineWidth = 1; g.stroke();
  g.beginPath(); w.seen.forEach((p, k) => k ? g.lineTo(X(p), Y(p)) : g.moveTo(X(p), Y(p)));
  g.strokeStyle = css('--ruler'); g.lineWidth = 2; g.stroke();
  if (w.seen.length < 400){ g.fillStyle = css('--ruler'); for (const p of w.seen){ g.beginPath(); g.arc(X(p), Y(p), 3, 0, 7); g.fill(); } }
}
function drawChart(cur){
  const c = $('chart'), g = c.getContext('2d'), W = c.width, H = c.height, l = 92, b = 44;
  const lx0 = Math.log(RMAX * 1.25), lx1 = Math.log(RMIN * 0.55), ly0 = Math.log(SIDE * 2.8), ly1 = Math.log(SIDE * 15);   /* lengths run from 3 to 12.6 sides */
  const X = v => l + (Math.log(v) - lx0) / (lx1 - lx0) * (W - l - 12), Y = v => H - b - (Math.log(v) - ly0) / (ly1 - ly0) * (H - b - 12);
  g.clearRect(0, 0, W, H); g.fillStyle = css('--muted'); g.strokeStyle = css('--line'); g.font = '22px system-ui,sans-serif';   /* the canvas is drawn at 680 wide and shown at about 390 on a phone */
  for (const v of [1, 3, 10, 30, 100]){ g.beginPath(); g.moveTo(X(v), 12); g.lineTo(X(v), H - b); g.stroke(); g.fillText(`${v} km`, X(v) - 22, H - 12); }
  for (const v of [1000, 1500, 2000, 3000, 4000]){ g.beginPath(); g.moveTo(l, Y(v)); g.lineTo(W - 12, Y(v)); g.stroke(); g.fillText(v.toLocaleString('en-GB'), 8, Y(v) + 8); }
  g.fillText('coast, km', l + 8, 32); g.fillText('ruler → shorter', W - 190, H - b - 10);
  g.fillStyle = css('--shore'); for (const [r, L] of hist){ g.beginPath(); g.arc(X(r), Y(L), 4, 0, 7); g.fill(); }
  g.fillStyle = css('--ruler'); g.beginPath(); g.arc(X(cur[0]), Y(cur[1]), 6, 0, 7); g.fill();
}
const fromSlider = v => RMAX * Math.pow(RMIN / RMAX, v / 1000), toSlider = r => Math.round(1000 * Math.log(r / RMAX) / Math.log(RMIN / RMAX));
function show(rKm){
  const m = measure(rKm / SIDE), L = m.L * SIDE;
  if (!hist.some(([r]) => Math.abs(r - rKm) < 1e-9)) hist.push([rKm, L]);
  $('read').textContent = `Ruler ${rKm < 10 ? rKm.toFixed(2) : rKm.toFixed(1)} km · ${m.steps} steps · coast ${Math.round(L).toLocaleString('en-GB')} km`;
  drawMap(m); drawChart([rKm, L]);
  window.coast = {r: rKm, steps: m.steps, L, hist: hist.length};
}
for (const k of [1, 2, 3, 4, 5]){ const r = SIDE / 3 ** k, b = document.createElement('button'); b.type = 'button'; b.dataset.k = k;
  b.textContent = `${r < 10 ? r.toFixed(2) : r.toFixed(1)} km`; b.onclick = () => { $('r').value = toSlider(r); show(r); }; $('presets').append(b); }
$('r').addEventListener('input', () => show(fromSlider(+$('r').value)));
matchMedia('(prefers-color-scheme: dark)').addEventListener('change', () => show(window.coast.r));
window.walkUnits = r => { const w = walk(r); return [w.steps, w.gap]; };
show(RMAX);
